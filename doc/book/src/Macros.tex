
%%% This defines quotation box
\newcommand{\warningbox}[1]
{ 
\indent\par\smallskip
   
\iflatexml
%Work for .html
   \color{red}

 \fcolorbox{red!70!white}{orange!10!white}{\parbox{.9\textwidth}{
   \textbf{Warning:\\
    \rmfamily}
{#1}
}
}
      \color{black}
\else
% Work for .PDF
\begin{tcolorbox}[float,
colback=red!10!white, colframe=black!10!red,coltitle=white,
title=\textbf{Warning:\\
    \rmfamily}]
	{#1}
\end{tcolorbox}
\fi
}

%%% This defines numeric panels
\newcounter{numericsctr}
\numberwithin{numericsctr}{section}
\newcommand{\numericspanel}[2]
{ 
\noindent
\refstepcounter{numericsctr}\par\smallskip
   
\iflatexml
%Work for .html
   \color{blue}

 \fcolorbox{blue!70!white}{green!10!white}{\parbox{.9\textwidth}{
   \textbf{Numerics panel~\thenumericsctr\\
    \rmfamily}
{#2}\label{#1}\hypertarget{#1}
}
}
      \color{black}
\else
% Work for .PDF
\begin{tcolorbox}[float=htpb!,breakable,
colback=blue!10!white, colframe=black!10!blue,coltitle=white,
title=\textbf{Numerics panel~\thenumericsctr\\
    \rmfamily}]
	{#2}\label{#1}
\end{tcolorbox}
\fi
}
