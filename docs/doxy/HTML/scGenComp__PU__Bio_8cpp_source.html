60// Adjusts the heartbeat time (integration step size)
61void scGenComp_PU_Bio::
62 Heartbeat_Adjust(void)
63{
64 /* This works with voltage step limitation
65 float diff_V = abs(m_Membrane_V-m_Membrane_V_last);
66 if(diff_V<0.5)
67 { // The membrane's voltage changed less than 0.5 mV; can increase
68 m_Heartbeat_time *=2;
69 m_dt = m_Heartbeat_time.to_seconds()*1000.;
70 DEBUG_SC_EVENT(name(),"Heartbeat set to " << m_dt );
71 }
72 else if(diff_V>2.0)
73 { // The membrane's voltage changed more than 2 mV; must decrease
74 m_Heartbeat_time /=2;
75 m_dt = m_Heartbeat_time.to_seconds()*1000.;
76 DEBUG_SC_EVENT(name(),"Heartbeat set to " << m_dt );
77 }
78
79 m_Membrane_V_last = m_Membrane_V; // Preserve the 'new last' potential
80*/
81
82 // This works with dV/dt limitation
83 float diff_V = abs(m_Membrane_dV);
84 if(diff_V<VOLTAGE_CHANGE_ALLOWED*2)
85 { // The membrane's voltage changed less than 0.5 mV; can increase
86 if(m_Heartbeat_time <s_Heartbeat_time_default*5)
87 {
88 HeartbeatTime_Set(m_Heartbeat_time *2);
89 DEBUG_SC_EVENT(name(),"Heartbeat set to " << m_dt );
90 }
91 }
92 if(diff_V>VOLTAGE_CHANGE_ALLOWED)
93 { // The membrane's voltage changed more than allowed; must decrease if above resolution
94 if((m_Heartbeat_time > s_Heartbeat_time_resolution))
95 {
96 HeartbeatTime_Set(m_Heartbeat_time /2);
97 DEBUG_SC_EVENT(name(),"Heartbeat set to " << m_dt );
98 }
99 }
