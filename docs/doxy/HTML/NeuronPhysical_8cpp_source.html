218void NeuronPhysical::
219 Calculate_Do()
220{
221 m_Input_dVdt = 0; // Maybe several input gradient contributions exist
222 m_dt = m_Heartbeat_time.to_seconds()*1000; // We calculate in msec
223 m_Membrane_V_Rushin = 0;
224 // From the previous iteration
225 m_Membrane_Last_dVdt = m_Membrane_dVdt_Resulting;
226 m_AIS_I = m_Membrane_V/MembraneResistanceGOhm_Get()/1000; // The AIS current, in pA
227 m_Membrane_dVdt_AIS = m_Membrane_V/MembraneTauMSec_Get()*1000; // The AIS gradient, in [V/s]
228 // Independently from the stage, the rush-in current contributes
229 if(m_RushinCurrent)
230 {
231 m_Membrane_dVdt_Rushin = m_RushinCurrent->VoltageGradient_Get(m_t);
232 }
233 else
234 {
235 m_Membrane_dVdt_Rushin = 0;
236 }
237
238 // The synaptic inputs are open only in the stage 'Computing'
239 // The rush-in input is open only in stage 'Relaxing'
240 // (the rush-in current is created at the end of 'Computing'
241 // Check if we do have rush-in current at all
242 // No synaptic current is assumed
243 switch(StageFlag_Get())
244 {
245 case GenCompStageMachine_t::gcsm_Relaxing:
246 {
247 break;
248 }
249 case GenCompStageMachine_t::gcsm_Computing:
250 {
251 // Previous membrane!
252 // Question: if the synaptic charge conserves
253 for(uint32_t i = 0; i< m_SynapticCurrents.size(); i++)
254 m_Input_dVdt += m_SynapticCurrents[i]->VoltageGradient_Get(m_t); // [V/s]
255 break;
256 }
257 case GenCompStageMachine_t::gcsm_Delivering:
258 {
259 break;
260 }
261 default: assert(0); break;
262 }
263
264 // The resulting gradient, in [mV/s]
265 m_Membrane_dVdt_Resulting = m_Membrane_dVdt_Rushin
266 + m_Input_dVdt
267 - m_Membrane_dVdt_AIS; // in [V/s]
268 // The Na+ current
269 m_Na_I += m_Membrane_dVdt_Rushin/MembraneResistanceGOhm_Get()/1000/1000;
270
271 m_MembraneGradientPositive =
272 (m_Membrane_dVdt_Resulting>0);
273
274 m_Membrane_dV = m_Membrane_dVdt_Resulting * m_dt; // The voltage change, in [mV], m_dt in [sec]
275
276 m_Membrane_V += m_Membrane_dV; // in [mV]
277 m_Resulting_I = - m_AIS_I + m_Na_I;
278 // Now we know all changed quantities; adjust the step size
279 Heartbeat_Adjust();
280}

310// return true if to stop heartbeating in 'Computing' mode
311bool NeuronPhysical::
312 Heartbeat_Computing_Stop()
313{ if(m_Membrane_V < 0)
314 { // Charging failed
315 ChargeupFailed();
316 return true;
317 }
318 else //No obvious problem, continue computing
319 {
320 if(m_Membrane_V < ThresholdPotential) return false;
321 // We are over, finish computing and start delivering
322 Create_Rushin();
323 return true;
324 }
325}

327// return true if to stop heartbeating in 'Delivering' mode
328bool NeuronPhysical::
329 Heartbeat_Delivering_Stop()
330{
331 if(m_Membrane_V >= ThresholdPotential)
332 return false; // Continue delivering
333 m_Relaxing_Stopped = false; // Relaxing just began
334 return true;
335}

337// return true if to stop heartbeating in 'Relaxing' mode
338bool NeuronPhysical::
339 Heartbeat_Relaxing_Stop()
340{
341 // Maybe we were relaxing and received our first input
342 // We are after delivering, check if before the hyperrelaxation turning point
343 if(!m_MembraneGradientPositive) return false;
344 // We are over the hyperpolarization turning point, check if resting
345 if ((abs(m_Membrane_V )>= AllowedRestingPotentialDifference))
346 return false; // Continue relaxing
347 // Make sure the resting potential really zero
348 m_Membrane_V = 0; m_Relaxing_Stopped = true;
349 // Cancel the synaptic inputs currents
350 while(m_SynapticCurrents.size())
351 InputCurrentDelete(m_SynapticCurrents[0]);
352 // Cancel rush-in current
353 delete m_RushinCurrent;
354 m_RushinCurrent = (NeuronInputCurrent*)NULL;
355 return true;
356}
